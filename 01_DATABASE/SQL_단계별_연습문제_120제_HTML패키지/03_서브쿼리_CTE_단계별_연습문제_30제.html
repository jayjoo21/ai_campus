    SELECT customer_id, COUNT(*) AS account_count, SUM(balance) AS total_balance

    GROUP BY customer_id
),

    SELECT customer_id, COUNT(*) AS transaction_count, SUM(amount) AS total_transaction_amount

Leakage Validation
문제
시점조건을 적용한 후보에서 opened_at이 assigned_at보다 늦은 행이 있는지 확인하세요.
사용 스키마fintech_fraud_lab_class
기대 결과leakage_rows=0이어야 합니다.
결과 Grain검증 결과 1행
작성 전에 생각할 것어느 테이블에서 시작하고 어떤 조건·그룹·순서를 적용할지 먼저 말로 정리합니다.
힌트 보기
사용 테이블: fintech_fraud_lab_class.campaign_experiments, fintech_fraud_lab_class.accounts
정상 JOIN 조건과 반대조건을 검증부에서 확인합니다.
정답과 해설 보기
USE fintech_fraud_lab_class;
WITH candidates AS (
    SELECT ce.experiment_id, ce.assigned_at, a.account_id, a.opened_at
    FROM campaign_experiments AS ce
    LEFT JOIN accounts AS a
      ON a.customer_id=ce.customer_id
     AND a.opened_at<=ce.assigned_at
)
SELECT COUNT(*) AS leakage_rows
FROM candidates
WHERE opened_at > assigned_at;
해설
정상적으로 시점조건이 적용됐다면 미래계좌는 0건이어야 합니다.

힌트 보기

USE fintech_fraud_lab_class;
WITH period_payments AS (
    SELECT payment_id, customer_id, amount
    FROM card_payments
    WHERE is_approved=1
      AND payment_at>='2026-01-01'
      AND payment_at<'2026-08-01'
),
customer_fds_summary AS (
    SELECT customer_id,
           COUNT(*) AS payment_count,
           SUM(amount) AS total_amount,
           AVG(amount) AS avg_amount
    FROM period_payments
    GROUP BY customer_id
)
SELECT *
FROM customer_fds_summary
ORDER BY total_amount DESC
LIMIT 100;

계좌·대출·거래를 세 사전집계 CTE로 만든 뒤 고객에 붙이세요.
사용 스키마fintech_fraud_lab_class
기대 결과고객마다 세 영역 요약이 한 행에 보입니다.
결과 Graincustomer_id 1명 = 1행
작성 전에 생각할 것어느 테이블에서 시작하고 어떤 조건·그룹·순서를 적용할지 먼저 말로 정리합니다.
힌트 보기
사용 테이블: fintech_fraud_lab_class.accounts, fintech_fraud_lab_class.loans, fintech_fraud_lab_class.transactions, fintech_fraud_lab_class.customers
각 CTE가 customer_id당 1행인지 확인합니다.
정답과 해설 보기
USE fintech_fraud_lab_class;
WITH acct AS (
    SELECT customer_id, COUNT(*) AS account_count, SUM(balance) AS total_balance
    FROM accounts GROUP BY customer_id
),
loan AS (
    SELECT customer_id, COUNT(*) AS loan_count, SUM(principal) AS total_principal
    FROM loans GROUP BY customer_id
),
txn AS (
    SELECT customer_id, COUNT(*) AS transaction_count, SUM(amount) AS total_transaction_amount
    FROM transactions GROUP BY customer_id
)
SELECT c.customer_id, c.region,
       COALESCE(acct.account_count,0) AS account_count,
       COALESCE(loan.loan_count,0) AS loan_count,
       COALESCE(txn.transaction_count,0) AS transaction_count
FROM customers AS c
LEFT JOIN acct ON acct.customer_id=c.customer_id
LEFT JOIN loan ON loan.customer_id=c.customer_id
LEFT JOIN txn ON txn.customer_id=c.customer_id;
해설
시점이 필요한 프로젝트에서는 현재상태가 아니라 기준시점 Snapshot으로 발전시켜야 합니다.

프로젝트형 Customer 360 Snapshot
프로젝트Point-in-TimeCustomer 360
문제
캠페인 배정시점 계좌·대출·90일 거래·동의상태를 experiment 한 행에 결합하세요.
사용 스키마fintech_fraud_lab_class
기대 결과experiment_id별 account_count·loan_count·originated_principal_sum·최근 90일 거래·AI_PROFILING 동의 여부가 한 행에 보입니다.
결과 Grainexperiment_id 1건 = 1행
작성 전에 생각할 것어느 테이블에서 시작하고 어떤 조건·그룹·순서를 적용할지 먼저 말로 정리합니다.
힌트 보기
사용 테이블: fintech_fraud_lab_class.campaign_experiments, fintech_fraud_lab_class.accounts, fintech_fraud_lab_class.loans, fintech_fraud_lab_class.transactions, fintech_fraud_lab_class.consent_records
모든 보조 CTE를 experiment_id당 1행으로 만듭니다. 계좌·대출은 opened_at <= assigned_at, 거래는 직전 90일, 동의는 granted_at/revoked_at/retention_days로 기준시점 유효성을 판정합니다.
정답과 해설 보기
USE fintech_fraud_lab_class;
WITH ce AS (
    SELECT experiment_id, customer_id, campaign_name, group_name,
           assigned_at, converted
    FROM campaign_experiments
),
acct AS (
    SELECT ce.experiment_id,
           COUNT(DISTINCT a.account_id) AS account_count
    FROM ce
    LEFT JOIN accounts AS a
      ON a.customer_id = ce.customer_id
     AND a.opened_at <= ce.assigned_at
    GROUP BY ce.experiment_id
),
loan AS (
    SELECT ce.experiment_id,
           COUNT(DISTINCT l.loan_id) AS loan_count,
           COALESCE(SUM(l.principal), 0) AS originated_principal_sum
    FROM ce
    LEFT JOIN loans AS l
      ON l.customer_id = ce.customer_id
     AND l.opened_at <= ce.assigned_at
    GROUP BY ce.experiment_id
),
txn AS (
    SELECT ce.experiment_id,
           COUNT(t.transaction_id) AS txn_90d,
           SUM(CASE
                 WHEN t.transaction_type IN ('DEPOSIT','TRANSFER_IN','INTEREST')
                 THEN t.amount ELSE 0
               END) AS inflow_90d,
           SUM(CASE
                 WHEN t.transaction_type IN ('WITHDRAWAL','TRANSFER_OUT','FEE')
                 THEN t.amount ELSE 0
               END) AS outflow_90d
    FROM ce
    LEFT JOIN transactions AS t
      ON t.customer_id = ce.customer_id
     AND t.transaction_at < ce.assigned_at
     AND t.transaction_at >= DATE_SUB(ce.assigned_at, INTERVAL 90 DAY)
    GROUP BY ce.experiment_id
),
consent AS (
    SELECT ce.experiment_id,
           MAX(
               CASE
                 WHEN cr.consent_type = 'AI_PROFILING'
                  AND cr.granted_at <= ce.assigned_at
                  AND (cr.revoked_at IS NULL OR cr.revoked_at > ce.assigned_at)
                  AND DATE_ADD(
                        cr.granted_at,
                        INTERVAL cr.retention_days DAY
                      ) > ce.assigned_at
                 THEN 1
                 ELSE 0
               END
           ) AS ai_profiling_granted
    FROM ce
    LEFT JOIN consent_records AS cr
      ON cr.customer_id = ce.customer_id
    GROUP BY ce.experiment_id
)
SELECT ce.experiment_id, ce.customer_id, ce.campaign_name, ce.group_name,
       ce.assigned_at, ce.converted,
       COALESCE(acct.account_count, 0) AS account_count,
       COALESCE(loan.loan_count, 0) AS loan_count,
       COALESCE(loan.originated_principal_sum, 0) AS originated_principal_sum,
       COALESCE(txn.txn_90d, 0) AS txn_90d,
       COALESCE(txn.inflow_90d, 0) AS inflow_90d,
       COALESCE(txn.outflow_90d, 0) AS outflow_90d,
       COALESCE(consent.ai_profiling_granted, 0) AS ai_profiling_granted
FROM ce
LEFT JOIN acct
  ON acct.experiment_id = ce.experiment_id
LEFT JOIN loan
  ON loan.experiment_id = ce.experiment_id
LEFT JOIN txn
  ON txn.experiment_id = ce.experiment_id
LEFT JOIN consent
  ON consent.experiment_id = ce.experiment_id
ORDER BY ce.assigned_at, ce.experiment_id;
해설
최종 Snapshot은 모든 보조 CTE를 experiment_id 1행으로 맞춰 결합합니다. 계좌·대출은 opened_at으로 기준시점 이전 존재 여부만 사용하고, 현재 상태인 balance/status/overdue_days는 과거 Feature로 쓰지 않습니다. 거래는 배정 직전 90일만 사용하고, 동의는 부여·철회·보유기간으로 당시 유효성을 재구성합니다. converted는 배정 이후의 성과이므로 Feature가 아니라 Target/성과 컬럼으로 분리해 해석해야 합니다.
